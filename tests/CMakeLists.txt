# Where tests that must write a file put it: inside the build tree, so it
# goes with the build directory. See tests/scratch.h for why not /tmp.
set(MABUR_TEST_SCRATCH_DIR ${CMAKE_CURRENT_BINARY_DIR}/scratch)
file(MAKE_DIRECTORY ${MABUR_TEST_SCRATCH_DIR})

function(add_mtest name)
  cmake_parse_arguments(MT "" "" "LIBS" ${ARGN})
  add_executable(${name} ${MT_UNPARSED_ARGUMENTS})
  target_link_libraries(${name} PRIVATE mabur_common ${MT_LIBS})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/third_party)
  target_compile_definitions(${name} PRIVATE
    MABUR_VECTOR_DIR="${CMAKE_CURRENT_SOURCE_DIR}/vectors"
    MABUR_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures"
    MABUR_TEST_SCRATCH_DIR="${MABUR_TEST_SCRATCH_DIR}")
  add_test(NAME ${name} COMMAND ${name})
endfunction()

# --- synthetic .gfont fixtures ----------------------------------------
#
# gen_gsfont.py is a real Python process: interpreter startup plus a
# pure-Python box blur over every requested size. Generating one per TEST
# (~1.8 s each) dominated a file's wall clock, and caching per binary still
# had test_gs_overlay and test_osd_coexist each spending ~11 s building the
# SAME 30-size set. They are built here instead: once per distinct size set,
# at build time, into the build tree. Shared across binaries, regenerated
# when the generator changes, removed with the build directory -- and,
# because nothing is generated by the first test to ask for it, a parallel
# ctest cannot have two processes writing one path.
set(GSFONT_DIR ${CMAKE_CURRENT_BINARY_DIR}/gsfonts)
set(GSFONT_OUTPUTS "")
function(add_gsfont var name sizes)
  set(out ${GSFONT_DIR}/${name}.gfont)
  add_custom_command(
    OUTPUT ${out}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${GSFONT_DIR}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/msp/gen_gsfont.py
            --synthetic ${out} --sizes ${sizes}
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/msp/gen_gsfont.py
    COMMENT "Generating synthetic ${name}.gfont (sizes ${sizes})"
    VERBATIM)
  set(${var} ${out} PARENT_SCOPE)
  set(GSFONT_OUTPUTS ${GSFONT_OUTPUTS} ${out} PARENT_SCOPE)
endfunction()

# The 30-size bake set has ONE definition, in gs_scaled_sizes.h -- the header
# explains why three places already have to agree on it. Parsed out rather
# than retyped here, which would make a fourth.
file(READ ${CMAKE_CURRENT_SOURCE_DIR}/gs_scaled_sizes.h MABUR_GS_SIZES_HDR)
string(REGEX MATCH "kScaledSizes[^;]*;" MABUR_GS_SIZES_DECL "${MABUR_GS_SIZES_HDR}")
string(REGEX MATCHALL "\"[^\"]*\"" MABUR_GS_SIZES_PARTS "${MABUR_GS_SIZES_DECL}")
set(MABUR_GS_SCALED_SIZES "")
foreach(part IN LISTS MABUR_GS_SIZES_PARTS)
  string(REPLACE "\"" "" part "${part}")
  string(APPEND MABUR_GS_SCALED_SIZES "${part}")
endforeach()
if(NOT MABUR_GS_SCALED_SIZES MATCHES "^[0-9]+(,[0-9]+)*$")
  message(FATAL_ERROR
    "could not parse kScaledSizes out of tests/gs_scaled_sizes.h "
    "(got '${MABUR_GS_SCALED_SIZES}')")
endif()
